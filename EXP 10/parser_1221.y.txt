%{
#include <stdio.h>
#include <stdlib.h>

int yylex();
void yyerror(const char *s);
%}

%token FOR INT ID NUM
%token ASSIGN INC DEC
%token LE GE EQ NE LT GT
%token PLUS MINUS

%left PLUS MINUS

%%

program:
    for_loop { printf("Valid for loop!\n"); }
    ;

for_loop:
    FOR '(' initialization ';' condition ';' increment ')' statement
    ;

initialization:
    INT ID ASSIGN expression
    | ID ASSIGN expression
    | /* empty */
    ;

condition:
    expression relational_operator expression
    | /* empty */
    ;

relational_operator:
    LE | GE | EQ | NE | LT | GT
    ;

increment:
    ID INC
    | ID DEC
    | INC ID
    | DEC ID
    | ID ASSIGN expression
    | /* empty */
    ;

expression:
    ID
    | NUM
    | expression PLUS expression
    | expression MINUS expression
    ;

statement:
    ID ASSIGN expression ';'
    | ';'
    ;

%%

void yyerror(const char *s) {
    printf("Invalid for loop!\n");
}

int main() {
    printf("Enter a for loop:\n");
    yyparse();
    return 0;
}
