%{
#include "y.tab.h"
%}

%%

"for"                       { return FOR; }
"int"                       { return INT; }
[a-zA-Z_][a-zA-Z0-9_]*      { return ID; }
[0-9]+                      { return NUM; }
"="                         { return ASSIGN; }
"<="                        { return LE; }
">="                        { return GE; }
"=="                        { return EQ; }
"!="                        { return NE; }
"<"                         { return LT; }
">"                         { return GT; }
"++"                        { return INC; }
"--"                        { return DEC; }
"+"                         { return PLUS; }
"-"                         { return MINUS; }
"("                         { return '('; }
")"                         { return ')'; }
";"                         { return ';'; }
[\t\n]+                     ; /* Ignore whitespace and newlines */
.                           { return yytext[0]; }

%%

int yywrap() {
    return 1;
}
